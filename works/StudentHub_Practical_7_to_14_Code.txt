CREATE DATABASE IF NOT EXISTS studenthub
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;
USE studenthub;

CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(120) NOT NULL,
    student_id VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(190) NOT NULL UNIQUE,
    mobile VARCHAR(20) NOT NULL,
    course VARCHAR(120) NOT NULL,
    study_year TINYINT UNSIGNED NOT NULL,
    semester TINYINT UNSIGNED NOT NULL,
    gender VARCHAR(30) NOT NULL,
    password VARCHAR(255) NOT NULL,
    role ENUM('student', 'admin') NOT NULL DEFAULT 'student',
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS students (
    id INT AUTO_INCREMENT PRIMARY KEY,
    student_id VARCHAR(40) NOT NULL UNIQUE,
    full_name VARCHAR(120) NOT NULL,
    email VARCHAR(190) NOT NULL UNIQUE,
    course VARCHAR(120) NOT NULL,
    study_year TINYINT UNSIGNED NOT NULL,
    semester TINYINT UNSIGNED NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS events (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    description TEXT NOT NULL,
    event_date DATE NOT NULL,
    event_time VARCHAR(30) NOT NULL,
    venue VARCHAR(160) NOT NULL,
    category VARCHAR(80) NOT NULL,
    poster_path VARCHAR(255) DEFAULT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS registrations (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    event_id INT NOT NULL,
    registered_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY uq_registration (user_id, event_id),
    CONSTRAINT fk_registration_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT fk_registration_event FOREIGN KEY (event_id) REFERENCES events(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS remember_tokens (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    selector CHAR(24) NOT NULL UNIQUE,
    validator_hash CHAR(64) NOT NULL,
    expires_at DATETIME NOT NULL,
    CONSTRAINT fk_remember_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS audit_logs (
    id INT AUTO_INCREMENT PRIMARY KEY,
    actor_id INT DEFAULT NULL,
    action VARCHAR(80) NOT NULL,
    entity VARCHAR(80) NOT NULL,
    entity_id INT DEFAULT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_audit_created_at (created_at),
    CONSTRAINT fk_audit_actor FOREIGN KEY (actor_id) REFERENCES users(id) ON DELETE SET NULL
) ENGINE=InnoDB;

const SESSION_TIMEOUT_SECONDS = 1800;

function json_response(array $payload, int $status = 200): void
{
    http_response_code($status);
    header("Content-Type: application/json; charset=utf-8");
    echo json_encode($payload, JSON_UNESCAPED_SLASHES);
    exit;
}

function require_login(bool $json = false): void
{
    if (isset($_SESSION["user_id"])) {
        $lastActivity = (int)($_SESSION["last_activity"] ?? time());
        if (time() - $lastActivity > SESSION_TIMEOUT_SECONDS) {
            $_SESSION = [];
            session_destroy();
            if ($json) {
                json_response(["success" => false, "message" => "Session expired. Please log in again."], 401);
            }
            header("Location: login.html?expired=1");
            exit;
        }
        $_SESSION["last_activity"] = time();
        return;
    }

    if ($json) {
        json_response(["success" => false, "message" => "Please log in to continue."], 401);
    }
    header("Location: login.html");
    exit;
}

function require_admin(bool $json = false): void
{
    require_login($json);
    if (($_SESSION["role"] ?? "student") !== "admin") {
        if ($json) {
            json_response(["success" => false, "message" => "Administrator access is required."], 403);
        }
        http_response_code(403);
        exit("Administrator access is required.");
    }
}

function csrf_token(): string
{
    if (empty($_SESSION["csrf_token"])) {
        $_SESSION["csrf_token"] = bin2hex(random_bytes(32));
    }
    return $_SESSION["csrf_token"];
}

function require_csrf(): void
{
    $provided = $_SERVER["HTTP_X_CSRF_TOKEN"] ?? $_POST["csrf_token"] ?? "";
    if (!is_string($provided) || !hash_equals(csrf_token(), $provided)) {
        json_response(["success" => false, "message" => "Invalid request token. Refresh the page and try again."], 419);
    }
}

function clear_remember_cookie(): void
{
    setcookie("studenthub_remember", "", [
        "expires" => time() - 3600,
        "httponly" => true,
        "secure" => !empty($_SERVER["HTTPS"]) && $_SERVER["HTTPS"] !== "off",
        "samesite" => "Lax",
        "path" => "/",
    ]);
}

function restore_remembered_session(): void
{
    global $pdo;

    if (isset($_SESSION["user_id"]) || empty($_COOKIE["studenthub_remember"])) {
        return;
    }

    $parts = explode(":", $_COOKIE["studenthub_remember"], 2);
    if (count($parts) !== 2 || !preg_match('/^[a-f0-9]{24}$/', $parts[0]) || !preg_match('/^[a-f0-9]{64}$/', $parts[1])) {
        clear_remember_cookie();
        return;
    }

    [$selector, $validator] = $parts;
    $stmt = $pdo->prepare(
        "SELECT rt.id AS token_id, rt.user_id, rt.validator_hash, u.full_name, u.student_id, u.email, u.role
         FROM remember_tokens rt
         JOIN users u ON u.id = rt.user_id
         WHERE rt.selector = :selector AND rt.expires_at > NOW()
         LIMIT 1"
    );
    $stmt->execute([":selector" => $selector]);
    $record = $stmt->fetch();
    if (!$record || !hash_equals($record["validator_hash"], hash("sha256", $validator))) {
        if ($record) {
            $delete = $pdo->prepare("DELETE FROM remember_tokens WHERE id = :id");
            $delete->execute([":id" => $record["token_id"]]);
        }
        clear_remember_cookie();
        return;
    }

    session_regenerate_id(true);
    $_SESSION["user_id"] = (int)$record["user_id"];
    $_SESSION["student_id"] = $record["student_id"];
    $_SESSION["user_name"] = $record["full_name"];
    $_SESSION["user_email"] = $record["email"];
    $_SESSION["role"] = $record["role"];
    $_SESSION["last_activity"] = time();
}

restore_remembered_session();

    $course = isset($_GET["course"]) && is_string($_GET["course"]) ? trim($_GET["course"]) : "";
    $sql = "SELECT id, student_id, full_name, email, course, study_year, semester FROM students
            WHERE (? = '' OR full_name LIKE CONCAT('%', ?, '%') OR student_id LIKE CONCAT('%', ?, '%') OR email LIKE CONCAT('%', ?, '%'))
            AND (? = '' OR course = ?)
            ORDER BY created_at DESC";
    $stmt = $mysqli->prepare($sql);
    $stmt->bind_param("ssssss", $search, $search, $search, $search, $course, $course);
    $stmt->execute();
    $result = $stmt->get_result();
    json_response(["success" => true, "students" => $result->fetch_all(MYSQLI_ASSOC)]);

    json_response(["success" => false, "message" => "Method not allowed."], 405);

$input = json_decode(file_get_contents("php://input"), true);

$action = $field("action");
$id = isset($input["id"]) && (is_string($input["id"]) || is_int($input["id"])) ? filter_var($input["id"], FILTER_VALIDATE_INT) : false;

try {
    $mysqli->begin_transaction();
    if ($action === "delete") {
        if (!$id || $id < 1) {
            json_response(["success" => false, "message" => "Select a valid student."], 422);
        }
        $stmt = $mysqli->prepare("DELETE FROM students WHERE id = ?");
        $stmt->bind_param("i", $id);
        $stmt->execute();
        if ($stmt->affected_rows !== 1) {
            json_response(["success" => false, "message" => "Student not found."], 404);
        }
    } elseif ($action === "create" || $action === "update") {
        $studentId = $field("student_id");
        $fullName = $field("full_name");
        $email = strtolower($field("email"));
        $course = $field("course");
        $year = filter_var($field("study_year"), FILTER_VALIDATE_INT);
        $semester = filter_var($field("semester"), FILTER_VALIDATE_INT);
        if (
            !preg_match("/^[A-Za-z0-9-]{2,40}$/", $studentId) ||
            !preg_match("/^[\p{L}][\p{L} .'-]{1,119}$/u", $fullName) ||
            !filter_var($email, FILTER_VALIDATE_EMAIL) || strlen($email) > 190 ||
            $course === "" || strlen($course) > 120 ||
            $year === false || $year < 1 || $year > 4 ||
            $semester === false || $semester < 1 || $semester > 8 ||
            ($action === "update" && (!$id || $id < 1))
        ) {
            json_response(["success" => false, "message" => "Enter valid student details."], 422);
        }

        if ($action === "create") {
            $stmt = $mysqli->prepare(
                "INSERT INTO students (student_id, full_name, email, course, study_year, semester) VALUES (?, ?, ?, ?, ?, ?)"
            );
            $stmt->bind_param("ssssii", $studentId, $fullName, $email, $course, $year, $semester);
            $stmt->execute();
            $entityId = $stmt->insert_id;
        } else {
            $stmt = $mysqli->prepare(
                "UPDATE students SET student_id = ?, full_name = ?, email = ?, course = ?, study_year = ?, semester = ? WHERE id = ?"
            );
            $stmt->bind_param("ssssiii", $studentId, $fullName, $email, $course, $year, $semester, $id);
            $stmt->execute();
            if ($stmt->affected_rows === 0) {
                $check = $mysqli->prepare("SELECT id FROM students WHERE id = ?");
                $check->bind_param("i", $id);
                $check->execute();
                if (!$check->get_result()->fetch_assoc()) {
                    json_response(["success" => false, "message" => "Student not found."], 404);
                }
            }
            $entityId = $id;
        }
        $auditAction = $action === "create" ? "student.created" : "student.updated";
        $stmt = $mysqli->prepare("INSERT INTO audit_logs (actor_id, action, entity, entity_id) VALUES (?, ?, 'student', ?)");
        $actorId = (int)$_SESSION["user_id"];
        $stmt->bind_param("isi", $actorId, $auditAction, $entityId);
        $stmt->execute();
    } else {
        json_response(["success" => false, "message" => "Unknown student operation."], 400);
    }

    if ($action === "delete") {
        $auditAction = "student.deleted";
        $stmt = $mysqli->prepare("INSERT INTO audit_logs (actor_id, action, entity, entity_id) VALUES (?, ?, ?, ?)");
        $actorId = (int)$_SESSION["user_id"];
        $stmt->bind_param("issi", $actorId, $auditAction, $resource, $id);
        $stmt->execute();
    }
    $mysqli->commit();
    json_response(["success" => true, "message" => "Student " . ($action === "delete" ? "deleted" : ($action === "create" ? "added" : "updated")) . " successfully."]);
} catch (mysqli_sql_exception $e) {
    $mysqli->rollback();
    if ($e->getCode() === 1062) {
        json_response(["success" => false, "message" => "Student ID or email is already in use."], 409);
    }
    error_log("StudentHub student operation failed: " . $e->getMessage());
    json_response(["success" => false, "message" => "Student operation failed."], 500);
}

document.addEventListener("DOMContentLoaded", async () => {

        return;
    }

    const pageSize = 3;
    let events = [];
    let currentPage = 1;

    function createMeta(label, value) {
        const paragraph = document.createElement("p");
        const strong = document.createElement("strong");
        strong.textContent = `${label}: `;
        paragraph.append(strong, document.createTextNode(value));
        return paragraph;
    }

    function getVisibleEvents() {
        const searchTerm = searchInput.value.trim().toLocaleLowerCase();
        const selectedCategory = categorySelect.value;
        const sortedEvents = events
            .filter((item) => selectedCategory === "all" || item.category === selectedCategory)
            .filter((item) =>
                [item.title, item.description, item.venue, item.category]
                    .some((value) => value.toLocaleLowerCase().includes(searchTerm))
            );

        sortedEvents.sort((first, second) => {
            switch (sortSelect.value) {
                case "date-desc":
                    return second.date.localeCompare(first.date);
                case "title-asc":
                    return first.title.localeCompare(second.title);
                case "title-desc":
                    return second.title.localeCompare(first.title);
                default:
                    return first.date.localeCompare(second.date);
            }
        });

        return sortedEvents;
    }

    function render() {
        const visibleEvents = getVisibleEvents();
        const pageCount = Math.ceil(visibleEvents.length / pageSize);
        currentPage = Math.min(currentPage, Math.max(pageCount, 1));
        const pageEvents = visibleEvents.slice((currentPage - 1) * pageSize, currentPage * pageSize);

        eventList.replaceChildren();
        pagination.replaceChildren();

        if (visibleEvents.length === 0) {
            eventStatus.textContent = "No events match your search. Try another keyword or category.";
            return;
        }

        eventStatus.textContent = `Showing ${pageEvents.length} of ${visibleEvents.length} events.`;

        pageEvents.forEach((item) => {
            const article = document.createElement("article");
            article.className = "event-card";

            const title = document.createElement("h3");
            title.textContent = item.title;

            const description = document.createElement("p");
            description.textContent = item.description;

            const category = document.createElement("span");
            category.className = "event-category";
            category.textContent = item.category;

            const meta = document.createElement("div");

                    day: "numeric",
                    month: "long",
                    year: "numeric"
                }).format(new Date(`${item.date}T00:00:00`))),
                createMeta("Time", item.time),
                createMeta("Venue", item.venue)
            );

            article.append(title, category, description, meta);
            if (item.poster_path) {
                const poster = document.createElement("img");
                poster.src = item.poster_path;
                poster.alt = `${item.title} poster`;
                poster.loading = "lazy";
                article.prepend(poster);
            }
            const registerButton = document.createElement("button");
            registerButton.type = "button";
            registerButton.className = "secondary-button";
            registerButton.textContent = "Register for event";
            registerButton.addEventListener("click", async () => {
                registerButton.disabled = true;
                try {
                    const tokenResponse = await fetch("php/csrf.php");
                    const tokenResult = await tokenResponse.json();
                    if (!tokenResponse.ok || !tokenResult.token) {
                        throw new Error(tokenResult.message || "Log in before registering for an event.");
                    }
                    const response = await fetch("php/event_registration.php", {
                        method: "POST",
                        headers: { "Content-Type": "application/json", "X-CSRF-Token": tokenResult.token },
                        body: JSON.stringify({ event_id: item.id }),
                    });
                    const result = await response.json();
                    if (!response.ok || !result.success) throw new Error(result.message || "Registration failed.");
                    eventStatus.textContent = result.message;
                    eventStatus.classList.remove("error");
                } catch (error) {
                    eventStatus.textContent = error.message;

        for (let page = 1; page <= pageCount; page += 1) {
            const button = document.createElement("button");

            button.className = "pagination-button";
            button.textContent = String(page);

            button.disabled = page === currentPage;
            button.addEventListener("click", () => {
                currentPage = page;
                render();
            });
            pagination.append(button);
        }
    }

        if (!response.ok) {
            throw new Error(`Event data request failed with status ${response.status}.`);
        }

        if (!data.success || !Array.isArray(data.events)) {
            throw new Error(data.message || "Event data must be a JSON array.");
        }

        events = data.events.filter((item) =>
            item &&
            typeof item.id !== "undefined" &&
            typeof item.title === "string" &&
            typeof item.description === "string" &&
            typeof item.date === "string" &&
            typeof item.time === "string" &&
            typeof item.venue === "string" &&
            typeof item.category === "string"
        );

            const option = document.createElement("option");
            option.value = category;
            option.textContent = category;
            categorySelect.append(option);

            ) {

                showMessage(

   ========================================================= */
document.addEventListener("DOMContentLoaded", function () {
    const messageIds = [
        "contactMessage",
        "feedbackMessage",
        "profileMessage",
        "loginMessage",
        "registerMessage"
    ];

    messageIds.forEach(function (id) {
        const el = document.getElementById(id);
        if (!el) return;

        const observer = new MutationObserver(function () {
            const text = (el.textContent || "").toLowerCase();

            el.classList.remove("success", "error");

            if (!text.trim()) return;

            const isError =
                text.includes("error") ||
                text.includes("invalid") ||
                text.includes("fail") ||
                text.includes("incorrect") ||
                text.includes("required") ||
                text.includes("mismatch") ||
                text.includes("wrong");

            el.classList.add(isError ? "error" : "success");
        });

        observer.observe(el, {
            childList: true,
            characterData: true,
            subtree: true
        });
    });
});

# StudentHub Practicals 7–14

## Local setup

1. Start Apache and MySQL in XAMPP.
2. Import [`schema.sql`](./schema.sql) in phpMyAdmin or run it with the MySQL command line. It creates the `studenthub` database and all tables. If the database already contains the earlier Practical 3–6 tables, run [`migrate-existing.sql`](./migrate-existing.sql) once afterward to add the new user/event fields without losing existing records.
3. If needed, set `STUDENTHUB_DB_HOST`, `STUDENTHUB_DB_NAME`, `STUDENTHUB_DB_USER`, and `STUDENTHUB_DB_PASSWORD`. Defaults are `localhost`, `studenthub`, `root`, and an empty password for a standard XAMPP installation.
4. Open the project through Apache (for example `http://localhost/WDF%20PRACTICALS%20-%20Copy/`); do not open the HTML files as `file://` URLs.
5. Register an account, then promote it to administrator in MySQL:

   ```sql
   UPDATE studenthub.users SET role = 'admin' WHERE email = 'your-email@example.com';
   ```

6. Log in again. Admin accounts go to `admin.php`; students go to `profile.php`.

The web server must be able to write to `data/` for contact JSON submissions and to `uploads/event-posters/` for event poster uploads. Contact records are stored in `data/contact_submissions.json`; the included Apache rule prevents direct web access to that file.

## ER model

- **users** 1—many **registrations** many—1 **events**
- **students** stores the administrator-managed student directory independently from login accounts.
- **users** 1—many **remember_tokens**; tokens are selector/validator credentials and only validator hashes are stored.
- **users** 1—many **audit_logs** as the acting administrator.

`registrations` has a unique `(user_id, event_id)` pair and foreign keys to both parent records. Events and registration history are removed consistently when their parent is deleted.

## Practical feature map

- **7:** `contact.html` → `php/contact.php` validates form values and appends JSON records with an exclusive file lock.
- **8:** `schema.sql` and `php/config.php` provide the schema, PDO connection, and MySQLi connection.
- Event attendees can register from the public event list; registrations are unique per student and event.
- **9:** `register.html` → `php/register.php` validates fields, enforces unique email/Student ID, and stores `password_hash()` output using a MySQLi prepared statement.
- **10:** `php/login.php`, `php/logout.php`, `php/auth.php`, and `profile.php` provide password verification, regenerated sessions, role-aware redirects, a 30-minute idle timeout, and optional 30-day remember-me tokens.
- **11:** `students.php` and `php/students_api.php` implement admin-only MySQLi CRUD, search, course filtering, and feedback messages.
- **12:** `events_admin.php` and `php/events_api.php` implement admin-only event CRUD and JPEG/PNG/WebP poster validation (2 MB maximum).
- **13:** `js/management.js` uses Fetch for asynchronous student and event CRUD; the public `event.html` list loads from the JSON event API.
- **14:** `admin.php` provides admin-only counts, recent registrations, dynamic administrator navigation, and recent audit records.

All management API mutations require an authenticated administrator and a session CSRF token. A registered student is not an administrator unless their role is explicitly promoted.
